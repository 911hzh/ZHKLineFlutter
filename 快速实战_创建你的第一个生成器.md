## 步骤 2: 创建简单的生成器脚本 (15 分钟)

创建文件: `tool/simple_generator.dart`

```dart
import 'dart:io';

/// 简单的代码生成器
/// 这个示例不使用 build_runner，而是直接解析和生成
void main() {
  print('🔍 开始扫描项目...');

  // 扫描 lib 目录下的所有 dart 文件
  final libDir = Directory('lib');
  final dartFiles = libDir
      .listSync(recursive: true)
      .whereType<File>()
      .where((f) => f.path.endsWith('.dart'))
      .toList();

  print('📁 找到 ${dartFiles.length} 个 Dart 文件');

  int generatedCount = 0;

  for (final file in dartFiles) {
    final content = file.readAsStringSync();

    // 检查是否包含 @AutoToString() 注解
    if (content.contains('@AutoToString()')) {
      print('✨ 处理文件: ${file.path}');

      // 生成代码
      final generatedCode = generateToString(content, file.path);

      if (generatedCode.isNotEmpty) {
        // 写入生成的文件
        final generatedFile = File(file.path.replaceAll('.dart', '.g.dart'));
        generatedFile.writeAsStringSync(generatedCode);

        print('   ✓ 生成: ${generatedFile.path}');
        generatedCount++;
      }
    }
  }

  print('\n🎉 完成！生成了 $generatedCount 个文件');
}

/// 生成 toString 方法
String generateToString(String sourceCode, String filePath) {
  final buffer = StringBuffer();

  // 提取类名和字段
  final classInfo = extractClassInfo(sourceCode);

  if (classInfo == null) {
    print('   ⚠️  警告: 无法解析类信息');
    return '';
  }

  buffer.writeln('// GENERATED CODE - DO NOT MODIFY BY HAND');
  buffer.writeln('// Generated from: $filePath');
  buffer.writeln();
  buffer.writeln("part of '${filePath.split('/').last}';");
  buffer.writeln();
  buffer.writeln('extension ${classInfo.className}ToStringExtension on ${classInfo.className} {');
  buffer.writeln('  String toGeneratedString() {');

  // 生成字段字符串
  final fieldsStr = classInfo.fields.map((f) => "'$f': \$$f").join(', ');

  buffer.writeln('    return \'${classInfo.className}($fieldsStr)\';');
  buffer.writeln('  }');
  buffer.writeln('}');

  return buffer.toString();
}

/// 提取类信息（简化版本，实际应用中应该使用 analyzer）
ClassInfo? extractClassInfo(String sourceCode) {
  // 查找类定义
  final classPattern = RegExp(r'class\s+(\w+)');
  final classMatch = classPattern.firstMatch(sourceCode);

  if (classMatch == null) return null;

  final className = classMatch.group(1)!;

  // 查找字段
  final fieldPattern = RegExp(r'final\s+\w+\s+(\w+);');
  final fields = fieldPattern
      .allMatches(sourceCode)
      .map((m) => m.group(1)!)
      .toList();

  return ClassInfo(className: className, fields: fields);
}

class ClassInfo {
  final String className;
  final List<String> fields;

  ClassInfo({required this.className, required this.fields});
}
```

创建文件: `tool/analyzer_generator.dart`

```dart
import 'dart:io';
import 'package:analyzer/dart/analysis/features.dart';
import 'package:analyzer/dart/analysis/utilities.dart';
import 'package:analyzer/dart/ast/ast.dart';
import 'package:analyzer/dart/ast/visitor.dart';

/// 使用 Analyzer 的专业版本
void main() async {
  print('🔍 使用 Analyzer 扫描项目...');

  final libDir = Directory('lib');
  final dartFiles = libDir
      .listSync(recursive: true)
      .whereType<File>()
      .where((f) => f.path.endsWith('.dart') && !f.path.endsWith('.g.dart'))
      .toList();

  int generatedCount = 0;

  for (final file in dartFiles) {
    final content = file.readAsStringSync();

    if (!content.contains('@AutoToString')) continue;

    print('✨ 分析文件: ${file.path}');

    // 使用 Analyzer 解析
    final parseResult = parseString(
      content: content,
      featureSet: FeatureSet.latestLanguageVersion(),
      path: file.path,
    );

    final visitor = AutoToStringVisitor();
    parseResult.unit.accept(visitor);

    if (visitor.classesToGenerate.isNotEmpty) {
      final generatedCode = generateCodeFromAst(
        visitor.classesToGenerate,
        file.path,
      );

      final generatedFile = File(file.path.replaceAll('.dart', '.g.dart'));
      generatedFile.writeAsStringSync(generatedCode);

      print('   ✓ 生成: ${generatedFile.path}');
      generatedCount++;
    }
  }

  print('\n🎉 完成！生成了 $generatedCount 个文件');
}

/// AST Visitor - 遍历语法树
class AutoToStringVisitor extends RecursiveAstVisitor<void> {
  final List<ClassInfo> classesToGenerate = [];

  @override
  void visitClassDeclaration(ClassDeclaration node) {
    // 检查是否有 @AutoToString 注解
    final hasAnnotation = node.metadata.any(
      (annotation) => annotation.name.name == 'AutoToString',
    );

    if (hasAnnotation) {
      // 收集字段信息
      final fields = <FieldInfo>[];

      for (final member in node.members) {
        if (member is FieldDeclaration) {
          for (final variable in member.fields.variables) {
            fields.add(FieldInfo(
              name: variable.name.toString(),
              type: member.fields.type.toString(),
            ));
          }
        }
      }

      classesToGenerate.add(ClassInfo(
        className: node.name.toString(),
        fields: fields,
      ));

      print('   找到类: ${node.name} (${fields.length} 个字段)');
    }

    super.visitClassDeclaration(node);
  }
}

/// 从 AST 生成代码
String generateCodeFromAst(List<ClassInfo> classes, String sourcePath) {
  final buffer = StringBuffer();
  final fileName = sourcePath.split('/').last;

  buffer.writeln('// GENERATED CODE - DO NOT MODIFY BY HAND');
  buffer.writeln('// **************************************************************************');
  buffer.writeln('// AutoToStringGenerator');
  buffer.writeln('// **************************************************************************');
  buffer.writeln();
  buffer.writeln("part of '$fileName';");
  buffer.writeln();

  for (final classInfo in classes) {
    buffer.writeln('// Generated toString for ${classInfo.className}');
    buffer.writeln('extension ${classInfo.className}Extension on ${classInfo.className} {');
    buffer.writeln('  String toAutoString() {');

    // 生成字段列表
    final fieldList = classInfo.fields.map((f) => "'${f.name}': \$${f.name}").join(', ');

    buffer.writeln('    return \'${classInfo.className}($fieldList)\';');
    buffer.writeln('  }');
    buffer.writeln();

    // 生成 Map 方法
    buffer.writeln('  Map<String, dynamic> toMap() {');
    buffer.writeln('    return {');
    for (final field in classInfo.fields) {
      buffer.writeln("      '${field.name}': ${field.name},");
    }
    buffer.writeln('    };');
    buffer.writeln('  }');
    buffer.writeln('}');
    buffer.writeln();
  }

  return buffer.toString();
}

class ClassInfo {
  final String className;
  final List<FieldInfo> fields;

  ClassInfo({required this.className, required this.fields});
}

class FieldInfo {
  final String name;
  final String type;

  FieldInfo({required this.name, required this.type});
}
```

import 'package:k_line_flutter/annotations/auto_to_string.dart';

@AutoToString()
class User {
  final String name;
  final int age;
  final String email;

  User({
    required this.name,
    required this.age,
    required this.email,
  });

@AutoToString()
class Product {
  final String id;
  final String title;
  final double price;
  final bool inStock;

  Product({
    required this.id,
    required this.title,
    required this.price,
    required this.inStock,
  });

// 测试使用
void testGenerated() {
  final user = User(
    name: 'Alice',
    age: 25,
    email: 'alice@example.com',
  );

  // 使用生成的方法
  print(user.toAutoString());
  // 输出: User('name': Alice, 'age': 25, 'email': alice@example.com)

  print(user.toMap());
  // 输出: {name: Alice, age: 25, email: alice@example.com}

  final product = Product(
    id: 'P001',
    title: 'MacBook Pro',
    price: 2999.99,
    inStock: true,
  );

  print(product.toAutoString());
  // 输出: Product('id': P001, 'title': MacBook Pro, 'price': 2999.99, 'inStock': true)

## 🎓 进阶：集成到 build_runner

如果你想要完整的 build_runner 集成，创建一个独立的 Generator 包：

### 创建 `lib/generator/auto_to_string_generator.dart`

```dart
import 'package:analyzer/dart/element/element.dart';
import 'package:build/build.dart';
import 'package:source_gen/source_gen.dart';
import 'package:k_line_flutter/annotations/auto_to_string.dart';

class AutoToStringGenerator extends GeneratorForAnnotation<AutoToString> {
  @override
  String generateForAnnotatedElement(
    Element element,
    ConstantReader annotation,
    BuildStep buildStep,
  ) {
    if (element is! ClassElement) {
      throw InvalidGenerationSourceError(
        '@AutoToString 只能用于类',
        element: element,
      );
    }

    final className = element.name;
    final includeClassName = annotation.peek('includeClassName')?.boolValue ?? true;
    final includeFieldNames = annotation.peek('includeFieldNames')?.boolValue ?? true;

    // 获取所有字段
    final fields = element.fields.where((f) => !f.isStatic).toList();

    return _generateExtension(
      className: className,
      fields: fields,
      includeClassName: includeClassName,
      includeFieldNames: includeFieldNames,
    );
  }

  String _generateExtension({
    required String className,
    required List<FieldElement> fields,
    required bool includeClassName,
    required bool includeFieldNames,
  }) {
    final buffer = StringBuffer();

    buffer.writeln('extension ${className}Extension on $className {');
    buffer.writeln('  String toAutoString() {');

    if (fields.isEmpty) {
      buffer.writeln('    return \'$className()\';');
    } else {
      final fieldList = fields.map((f) {
        if (includeFieldNames) {
          return "'${f.name}': \$${f.name}";
        } else {
          return '\$${f.name}';
        }
      }).join(', ');

      if (includeClassName) {
        buffer.writeln('    return \'$className($fieldList)\';');
      } else {
        buffer.writeln('    return \'($fieldList)\';');
      }
    }

    buffer.writeln('  }');
    buffer.writeln('}');

    return buffer.toString();
  }
}
```

### 创建 Builder 文件: `lib/generator/builder.dart`

import 'package:build/build.dart';

## 📊 对比三种方法

| 方法              | 优点                 | 缺点                 | 适用场景   |
| ----------------- | -------------------- | -------------------- | ---------- |
| **简单脚本**      | 快速、易懂、无依赖   | 解析不准确、功能有限 | 学习、原型 |
| **Analyzer 脚本** | 准确、功能完整       | 需要手动运行         | 中小项目   |
| **build_runner**  | 自动化、集成好、标准 | 配置复杂、学习曲线   | 生产环境   |

## 🎯 总结

你现在学会了三种创建代码生成器的方法：

1. ✅ **简单脚本**: 字符串匹配 + 正则表达式
2. ✅ **Analyzer 脚本**: AST 分析 + Visitor 模式
3. ✅ **Build Runner**: 完整的代码生成框架

### 核心概念回顾：

- **注解 (Annotation)**: 标记需要生成代码的元素
- **AST (抽象语法树)**: 代码的结构化表示
- **Visitor 模式**: 遍历 AST 的方式
- **Code Generation**: 根据分析结果生成新代码

### Injectable Generator 使用的是方法 3！

它通过 build_runner + source_gen + analyzer 实现了：

- 🔍 扫描整个项目
- 📊 分析依赖关系
- 🔄 拓扑排序
- 📝 生成配置代码
